import pandas as pd import streamlit as st # Configuração da página st.set_page_config( page_title="Painel Financeiro - Raio-X", page_icon="💰", layout="wide" ) # Simulando uma base de dados inicial (substitua pela sua fonte real) if "dados_raio_x" not in st.session_state: st.session_state.dados_raio_x = pd.DataFrame([ { "id": 1, "categoria": "Manutenção carro", "parcela_atual": "08/12", "valor_parcela": 129.38, "nota": "", }, { "id": 2, "categoria": "Pessoal", "parcela_atual": "06/12", "valor_parcela": 288.90, "nota": "Tia Luciana", }, { "id": 3, "categoria": "Manutenção carro", "parcela_atual": "03/03", "valor_parcela": 140.00, "nota": "", }, { "id": 4, "categoria": "Pessoal", "parcela_atual": "02/02", "valor_parcela": 30.96, "nota": "", }, ]) st.title("📊 Painel de Controlo Financeiro") # --------------------------------------------------------- # CORREÇÃO 1: Cálculo dinâmico do Total de Gastos por Categoria # --------------------------------------------------------- st.subheader("TOTAL GASTOS POR CATEGORIA") df_atual = st.session_state.dados_raio_x # Agrupa e soma os valores do Raio-X por categoria automaticamente totais_por_categoria = ( df_atual.groupby("categoria")["valor_parcela"].sum().to_dict() ) # Lista completa de categorias padrão do sistema categorias_padrao = [ "Pessoal", "Mercado", "Combustível", "Saúde", "Manutenção Casa", "Manutenção carro", "Estudo", "Locomoção", "Lazer", "Presentes", ] # Exibição em formato de grelha (cartões) cols = st.columns(2) for i, cat in enumerate(categorias_padrao): valor_total = totais_por_categoria.get(cat, 0.0) with cols[i % 2]: st.markdown( f"""

{cat}

R$ {valor_total:,.2f}

""", unsafe_allow_html=True, ) st.markdown("---") # --------------------------------------------------------- # CORREÇÃO 2: Funcionalidade de Excluir Itens do Raio-X # --------------------------------------------------------- st.subheader("Raio-X de Compras e Parcelas Ativas") st.write("Somente compras reais e parcelas projetadas para este mês.") if df_atual.empty: st.info("Não existem itens ativos no Raio-X.") else: # Criar uma interface interativa para gerir e excluir linhas for index, row in df_atual.iterrows(): col_cat, col_parc, col_val, col_nota, col_acao = st.columns( [2, 1, 1, 2, 1] ) with col_cat: st.text(row["categoria"]) with col_parc: st.text(row["parcela_atual"]) with col_val: st.text(f"R$ {row['valor_parcela']:,.2f}") with col_nota: st.text(row["nota"] if row["nota"] else "-") with col_acao: # Botão de exclusão único para cada linha baseado no ID if st.button("🗑️ Excluir", key=f"del_{row['id']}"): st.session_state.dados_raio_x = df_atual.drop(index).reset_index( drop=True ) st.success(f"Item da categoria '{row['categoria']}' excluído!") st.rerun()
